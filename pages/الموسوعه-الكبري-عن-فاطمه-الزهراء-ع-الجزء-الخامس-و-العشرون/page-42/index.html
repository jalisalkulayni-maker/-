التحريم: 10/ ضَرَبَ اللَّهُ مَثَلًا لِلَّذِينَ كَفَرُوا امْرَأَتَ نُوحٍ وَ امْرَأَتَ لُوطٍ كانَتا تَحْتَ عَبْدَيْنِ مِنْ عِبادِنا صالِحَيْنِ فَخانَتاهُما فَلَمْ يُغْنِيا عَنْهُما مِنَ اللَّهِ شَيْئاً وَ قِيلَ ادْخُلَا النَّارَ مَعَ الدَّاخِلِينَ‏ ج 13/ 78، 153، 160، ج 19/ 145، ج 20/ 199، ج 21/ 193، 331، ج 21/ 331

التحريم: 11/ و ضَرَبَ اللَّهُ مَثَلًا لِلَّذِينَ كَفَرُوا امْرَأَتَ نُوحٍ وَ امْرَأَتَ لُوطٍ ج 13/ 361، ج 9/ 74، 75، ج 21/ 137، 264، ج 21/ 264

التحريم: 12/ وَ مَرْيَمَ ابْنَتَ عِمْرانَ الَّتِي أَحْصَنَتْ فَرْجَها فَنَفَخْنا فِيهِ مِنْ رُوحِنا ج 19/ 399، ج 22/ 19، 153

الملك: 27/ فَلَمَّا رَأَوْهُ زُلْفَةً سِيئَتْ وُجُوهُ الَّذِينَ كَفَرُوا وَ قِيلَ هذَا الَّذِي كُنْتُمْ بِهِ تَدَّعُونَ‏ ج 10/ 262، ج 24/ 83

القلم: 1- 6/ ن، وَ الْقَلَمِ، وَ ما يَسْطُرُونَ، ما أَنْتَ بِنِعْمَةِ رَبِّكَ بِمَجْنُونٍ، وَ إِنَّ لَكَ لَأَجْراً غَيْرَ مَمْنُونٍ، وَ إِنَّكَ لَعَلى‏ خُلُقٍ عَظِيمٍ، فَسَتُبْصِرُ وَ يُبْصِرُونَ، بِأَيِّكُمُ الْمَفْتُونُ‏ ج 9/ 195، ج 19/ 355، ج 21/ 377

القلم: 48/ فَاصْبِرْ لِحُكْمِ رَبِّكَ وَ لا تَكُنْ كَصاحِبِ الْحُوتِ‏ ج 7/ 221

الحاقّة: 28- 30/ ما أَغْنى‏ عَنِّي مالِيَهْ، هَلَكَ عَنِّي سُلْطانِيَهْ، خُذُوهُ فَغُلُّوهُ‏ ج 13/ 222، 295، ج 24/ 148

المعارج: 2/ سَأَلَ سائِلٌ بِعَذابٍ واقِعٍ، لِلْكافِرينَ لَيْسَ لَهُ دافِعٌ‏ ج 19/ 294، 311، ج 20/ 149، 175، ج 21/ 221، 308

نوح: 17/ وَ اللَّهُ أَنْبَتَكُمْ مِنَ الْأَرْضِ نَباتاً ج 13/ 313

نوح: 28/ رَبِّ اغْفِرْ لِي وَ لِوالِدَيَّ وَ لِمَنْ دَخَلَ بَيْتِيَ مُؤْمِناً وَ لِلْمُؤْمِنِينَ وَ الْمُؤْمِناتِ وَ لا تَزِدِ الظَّالِمِينَ إِلَّا تَباراً ج 21/ 306، ج 22/ 103

المزّمّل: 9/ رَبُّ الْمَشْرِقِ وَ الْمَغْرِبِ لا إِلهَ إِلَّا هُوَ فَاتَّخِذْهُ وَكِيلًا ج 9/ 77، 211، ج 22/ 14، 102

المدّثّر: 1/ يا أَيُّهَا الْمُدَّثِّرُ قُمْ فَأَنْذِرْ ج 24/ 204

المدّثّر: 28/ لا تُبْقِي وَ لا تَذَرُ ج 20/ 198

المدّثّر: 35/ إِنَّها لَإِحْدَى الْكُبَرِ نَذِيراً لِلْبَشَرِ ج 21/ 157، ج 22/ 14، 104

المدّثّر: 38/ كُلُّ نَفْسٍ بِما كَسَبَتْ رَهِينَةٌ ج 16/ 210

المدّثّر: 48/ فَما تَنْفَعُهُمْ شَفاعَةُ الشَّافِعِينَ‏ ج 10/ 262

الإنسان: 1- 32/ هَلْ أَتى‏ عَلَى الْإِنْسانِ حِينٌ مِنَ الدَّهْرِ ج 6/ 102، 103، ج 8/ 356، ج 19/ 270، ج 20/ 458، 464
