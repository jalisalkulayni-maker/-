الأحزاب: 37/ وَ إِذْ تَقُولُ لِلَّذِي أَنْعَمَ اللَّهُ عَلَيْهِ‏. أَمْسِكْ عَلَيْكَ زَوْجَكَ‏ ... ج 3/ 305، ج 21/ 331، 332، ج 22/ 167

الأحزاب: 40/ ما كانَ مُحَمَّدٌ أَبا أَحَدٍ مِنْ رِجالِكُمْ‏ ... ج 7/ 219، ج 18/ 136، 140

الأحزاب: 41/ اذْكُرُوا اللَّهَ ذِكْراً كَثِيراً ... ج 22/ 181، 200، 214، 247

الأحزاب: 53/ يا أَيُّهَا الَّذِينَ آمَنُوا لا تَدْخُلُوا بُيُوتَ النَّبِيِّ إِلَّا أَنْ يُؤْذَنَ لَكُمْ‏ ... وَ إِذا سَأَلْتُمُوهُنَّ مَتاعاً فَسْئَلُوهُنَّ مِنْ وَراءِ حِجابٍ‏ ... وَ ما كانَ لَكُمْ أَنْ تُؤْذُوا رَسُولَ اللَّهِ‏. ج 3/ 102، ج 10/ 320، 372، ج 12/ 255، 550، ج 13/ 82، 83، 145، ج 20/ 505، 517

الأحزاب: 56/ إِنَّ اللَّهَ وَ مَلائِكَتَهُ يُصَلُّونَ عَلَى النَّبِيِّ يا أَيُّهَا الَّذِينَ آمَنُوا صَلُّوا عَلَيْهِ وَ سَلِّمُوا تَسْلِيماً ... ج 8/ 308، ج 20/ 126، 293، ج 22/ 20، 162، 193، 212، 230، 252

الأحزاب: 57/ إِنَّ الَّذِينَ يُؤْذُونَ اللَّهَ وَ رَسُولَهُ لَعَنَهُمُ اللَّهُ فِي الدُّنْيا وَ الْآخِرَةِ وَ أَعَدَّ لَهُمْ عَذاباً مُهِيناً ... ج 10/ 199، ج 10/ 258، ج 12/ 197، 237، 333، 397، ج 13/ 23، 75، 111، 230، ج 15/ 264، ج 19/ 123، ج 20/ 215، 219، 221، 240، 241، 252، 274، 277، 283، ج 21/ 57، 61، ج 22/ 17

الأحزاب: 58/ وَ الَّذِينَ يُؤْذُونَ الْمُؤْمِنِينَ وَ الْمُؤْمِناتِ بِغَيْرِ مَا اكْتَسَبُوا فَقَدِ احْتَمَلُوا بُهْتاناً وَ إِثْماً مُبِيناً ..

ج 13/ 23، 58، ج 22/ 18، 143، 149

الأحزاب: 61- 62/ مَلْعُونِينَ أَيْنَما ثُقِفُوا أُخِذُوا وَ قُتِّلُوا تَقْتِيلًا سُنَّةَ اللَّهِ فِي الَّذِينَ خَلَوْا مِنْ قَبْلُ وَ لَنْ تَجِدَ لِسُنَّةِ اللَّهِ تَبْدِيلًا ... ج 7/ 195، 199، ج 20/ 217

الأحزاب: 64/ إِنَّ اللَّهَ لَعَنَ الْكافِرِينَ وَ أَعَدَّ لَهُمْ سَعِيراً ... ج 20/ 214

الأحزاب: 68/ ... وَ الْعَنْهُمْ لَعْناً كَبِيراً ... ج 20/ 215

الأحزاب: 72/ إِنَّا عَرَضْنَا الْأَمانَةَ عَلَى السَّماواتِ وَ الْأَرْضِ وَ الْجِبالِ فَأَبَيْنَ أَنْ يَحْمِلْنَها وَ أَشْفَقْنَ مِنْها وَ حَمَلَهَا الْإِنْسانُ إِنَّهُ كانَ ظَلُوماً جَهُولًا ... ج 1/ 218، ج 13/ 366

الأحزاب: 187/ يَسْئَلُونَكَ عَنِ السَّاعَةِ ... قُلْ إِنَّما عِلْمُها عِنْدَ رَبِّي لا يُجَلِّيها لِوَقْتِها إِلَّا هُوَ ثَقُلَتْ فِي السَّماواتِ وَ الْأَرْضِ لا تَأْتِيكُمْ إِلَّا بَغْتَةً يَسْئَلُونَكَ كَأَنَّكَ حَفِيٌّ عَنْها قُلْ إِنَّما عِلْمُها عِنْدَ اللَّهِ وَ لكِنَّ أَكْثَرَ النَّاسِ لا يَعْلَمُونَ‏ ... ج 24/ 87
