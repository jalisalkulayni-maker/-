468

عَنْ غَايَةِ وَصْفِهَا فَلَيْسَ لِأَحَدٍ أَنْ يَبْلُغَ شَيْئاً مِنْ وَصْفِكَ وَ يَعْرِفَ شَيْئاً مِنْ نَعْتِكَ إِلَّا مَا حَدَدْتَهُ وَ وَصَفْتَهُ وَ وَقَّفْتَهُ عَلَيْهِ وَ بَلَّغْتَهُ إِيَّاهُ فَأَنَا مُقِرٌّ بِأَنِّي لَا أَبْلُغُ مَا أَنْتَ أَهْلُهُ مِنْ تَعْظِيمِ جَلَالِكَ وَ تَقْدِيسِ مَجْدِكَ وَ تَمْجِيدِكَ وَ كَرَمِكَ وَ الثَّنَاءِ عَلَيْكَ وَ الْمَدْحِ لَكَ وَ الذِّكْرِ لِآلَائِكَ وَ الْحَمْدِ لَكَ عَلَى بَلَائِكَ وَ الشُّكْرِ لَكَ عَلَى نَعْمَائِكَ وَ ذَلِكَ مَا تَكِلُّ الْأَلْسُنُ عَنْ صِفَتِهِ وَ تَعْجِزُ الْأَبْدَانُ عَنْ أَدَاءِ شُكْرِهِ وَ إِقْرَارِي لَكَ بِمَا احْتَطَبْتُ عَلَى نَفْسِي مِنْ مُوبِقَاتِ الذُّنُوبِ الَّتِي قَدْ أَوْبَقَتْنِي وَ أَخْلَقَتْ عِنْدَكَ وَجْهِي وَ لِكَبِيرِ خَطِيئَتِي وَ عَظِيمِ جُرْمِي هَرَبْتُ إِلَيْكَ رَبِّي وَ جَلَسْتُ بَيْنَ يَدَيْكَ مَوْلَايَ وَ تَضَرَّعْتُ إِلَيْكَ سَيِّدِي لِأُقِرَّ لَكَ بِوَحْدَانِيَّتِكَ وَ بِوُجُودِ رُبُوبِيَّتِكَ فَأُثْنِي عَلَيْكَ بِمَا أَثْنَيْتَ عَلَى نَفْسِكَ وَ أَصِفُكَ بِمَا يَلِيقُ بِكَ مِنْ صِفَاتِكَ وَ أَذْكُرُ مَا أَنْعَمْتَ بِهِ عَلَيَّ مِنْ مَعْرِفَتِكَ وَ أَعْتَرِفُ لَكَ بِذُنُوبِي
